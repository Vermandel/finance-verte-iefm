\section{Model Equations (DICE-style)}

\subsection*{Time, Technology, Population}%

\begin{align}
t(i)  & = t_{0} + \Delta\, i,\\
A(i)  & = \frac{A(i-1)}{1 - g_{A} \exp(-\delta_{A} \Delta\, i)},\\
L(i)  & = L(i-1)\left( \frac{L_{\infty}}{L(i-1)}\right) ^{\ell_{g}}.
\end{align}


\subsection*{Carbon Intensity \& Abatement Cost Share}%

\begin{align}
\sigma(i)  & = \sigma(i-1)\,\exp\!\Big(-\,g_{\sigma}\,\big(1-\delta_{\sigma
}\big)^{\Delta(i-1)}\,\Delta\Big),\\
\theta_{1}(i)  & = \frac{p_{b}}{1000\,\theta_{2}}\,\big(1-\delta
_{pb}\big)^{i-1}\,\sigma(i).
\end{align}


\subsection*{Output, Damages/Abatement, Net Output}%

\begin{align}
Y(i)  & = A(i)\,K(i-1)^{\alpha}\,\Big(\tfrac{L(i)}{1000}\Big)^{1-\alpha},\\
Q(i)  & = \Delta\,\frac{\big(1-\theta_{1}(i)\,\mu(i)\big)^{\theta_{2}}}{1 +
a_{2}\,T_{AT}(i)^{a_{3}}}\,Y(i).
\end{align}


\subsection*{Consumption, Investment, Capital Accumulation}%

\begin{align}
C(i)  & = \big(1 - s(i)\big)\,Q(i),\\
K(i)  & = \big(1-\delta_{K}\big)^{\Delta}\,K(i-1) + s(i)\,Q(i).
\end{align}


\subsection*{Emissions \& Radiative Forcing}%

\begin{align}
E(i) &  =\Delta\Big[\sigma(i)\big(1-\mu(i)\big)Y(i)+E_{\text{Land}}(i)\Big],\\
E_{\text{Land}}(i) &  =E_{L0}\,\big(1-\delta_{EL}\big)^{i},\\
F_{EX}(i) &  =f_{0}+\min\!\Big(\tfrac{(f_{1}-f_{0})\,i}{t_{f}},\,f_{1}%
-f_{0}\Big),\\
F(i) &  =F_{2\times CO_{2}}\;\log_{2}\!\Big(\tfrac{M_{AT}(i-1)}{M_{AT0}%
}\Big)+F_{EX}(i).
\end{align}


\subsection*{Carbon Cycle (Three-Box)}%

\begin{align}
M_{AT}(i)  & = \big(1-\Delta b_{12}\big)\,M_{AT}(i-1) + \Delta b_{12}%
\,\frac{M_{AT0}}{M_{UP0}}\,M_{UP}(i-1) + \xi\,E(i),\\
M_{UP}(i)  & = \Delta b_{12}\,M_{AT}(i-1) + \Big(1-\Delta b_{12}\tfrac
{M_{AT0}}{M_{UP0}}-\Delta b_{23}\Big)\,M_{UP}(i-1) + \Delta b_{23}%
\,\frac{M_{UP0}}{M_{LO0}}\,M_{LO}(i-1),\\
M_{LO}(i)  & = \Delta b_{23}\,M_{UP}(i-1) + \Big(1-\Delta b_{23}\tfrac
{M_{UP0}}{M_{LO0}}\Big)\,M_{LO}(i-1).
\end{align}


\subsection*{Temperature Dynamics (Two-Box)}%

\begin{align}
T_{AT}(i) &  =T_{AT}(i-1)+\Delta c_{1}\,F(i)-\Delta c_{1}\,\frac{F_{2\times
CO_{2}}}{T_{2\times CO_{2}}}\,T_{AT}(i-1)-\Delta c_{1}c_{3}\big[T_{AT}%
(i-1)-T_{LO}(i-1)\big],\\
T_{LO}(i) &  =T_{LO}(i-1)+\Delta c_{4}\big[T_{AT}(i-1)-T_{LO}(i-1)\big].
\end{align}


\subsection*{Planner's Objective (CRRA over consumption per capita)}%

\begin{align}
c_{pc}(i)  & = 1000\,\frac{C(i)}{L(i)},\\
u(i)  & =
\begin{cases}
L(i)\,\dfrac{c_{pc}(i)^{1-\gamma}-1}{1-\gamma}, & \gamma\neq1,\\[6pt]%
L(i)\,\ln c_{pc}(i), & \gamma= 1,
\end{cases}
\\
W  & = \sum_{i \in\mathcal{I}} \beta(i)\,u(i), \quad\beta(i)=\Big(\tfrac
{1}{1+\rho}\Big)^{\Delta(i-i_{0})}.
\end{align}


\subsection*{Key Symbols}

\begin{itemize}
\item $\Delta$ time step; $\rho$ pure rate of time preference; $\gamma$ CRRA.

\item $A$ TFP; $K$ capital; $L$ population; $\alpha$ capital share.

\item $\sigma$ carbon intensity; $\mu$ abatement rate; $\theta_{1},\theta_{2}$
abatement cost params.

\item $a_{2},a_{3}$ damage function params; $T_{AT},T_{LO}$ temperatures
(atmosphere, deep ocean).

\item $E$ emissions; $E_{\text{Land}}$ land-use emissions; $F$ forcing;
$F_{EX}$ exogenous forcing.

\item Carbon boxes: $M_{AT},M_{UP},M_{LO}$ with preindustrial baselines
$M_{AT0},M_{UP0},M_{LO0}$.

\item Flows $b_{12},b_{23}$; conversion $\xi$; climate response $c_{1}%
,c_{3},c_{4}$; sensitivities $F_{2\times CO_{2}}$, $T_{2\times CO_{2}}$.
\end{itemize}
